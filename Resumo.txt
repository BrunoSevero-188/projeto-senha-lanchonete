// ==================== LCD 16x2 I2C ====================
// Endereços comuns: 0x27 (padrão), 0x3F (alternativo)
const uint8_t LCD_ADDRS[] = {0x27, 0x3F};
const uint8_t LCD_ADDR_COUNT = 2;
LiquidCrystal_I2C lcd(0x27, 16, 2);
uint8_t lcdEnderecoAtual = 0;
bool lcdIniciado = false;

bool iniciarLCD() {
  for (uint8_t i = 0; i < LCD_ADDR_COUNT; i++) {
    lcd = LiquidCrystal_I2C(LCD_ADDRS[i], 16, 2);
    lcd.init();
    lcd.backlight();
    lcd.clear();
    lcd.setCursor(0, 0);
    lcd.print("Teste LCD...");
    delay(500);
    // Se chegou aqui sem crash, assume que funcionou
    lcdEnderecoAtual = i;
    lcdIniciado = true;
    Serial.printf("LCD detectado no endereco 0x%02X\n", LCD_ADDRS[i]);
    return true;
  }
  Serial.println("ERRO: LCD nao encontrado em 0x27 nem 0x3F");
  return false;
}

const int MAX_SENHA = 9999;
const int MIN_SENHA = 1;

const int MAX_DIGITOS = 4;

const int EEPROM_SIZE = 512;
const int ADDR_SENHA_ATUAL = 0;
const int ADDR_TOTAL_HIST = 4;
const int ADDR_HISTORICO = 8;

// ==================== ESTRUTURAS ====================
struct RegistroSenha {
  int numero;
  char horario[9];  // HH:MM:SS
  bool valido;
};

struct EstadoSistema {
  int senhaAtual;
  RegistroSenha historico[LIMITE_HISTORICO];
  int totalHistorico;
};

EstadoSistema estado;

// Buffer de digitação
char bufferDigita[5] = {0};
int posDigita = 0;
bool modoConfirmacao = false;
bool modoManual = false;
unsigned long ultimaTecla = 0;

// ==================== WEBSOCKET ====================
void webSocketEvent(uint8_t num, WStype_t type, uint8_t* payload, size_t length) {
  if (type == WStype_CONNECTED) {
    IPAddress ip = webSocket.remoteIP(num);
    Serial.printf("[WS] Cliente %u conectado: %s\n", num, ip.toString().c_str());
    // Envia estado atual ao conectar
    JsonDocument doc;
    doc["senha"] = estado.senhaAtual > 0 ? String(estado.senhaAtual) : "---";
    doc["horario"] = obterHorario();
    doc["historico_count"] = estado.totalHistorico;
    String resp; serializeJson(doc, resp);
    webSocket.sendTXT(num, resp);
  } else if (type == WStype_DISCONNECTED) {
    Serial.printf("[WS] Cliente %u desconectado\n", num);
  }
}

// ==================== FUNÇÕES AUXILIARES ====================
String obterHorario() {
  // Usa tempo real se WiFi conectado, senão millis()
  if (WiFi.status() == WL_CONNECTED) {
    struct tm timeinfo;
    if (getLocalTime(&timeinfo)) {
      char buf[9];
      strftime(buf, sizeof(buf), "%H:%M:%S", &timeinfo);
      return String(buf);
    }
  }
  unsigned long segs = millis() / 1000;
  int h = (segs / 3600) % 24;
  int m = (segs % 3600) / 60;
  int s = segs % 60;
  char buf[9];
  sprintf(buf, "%02d:%02d:%02d", h, m, s);
  return String(buf);
}

  // 3 bips de 880Hz (igual ao web)
  for (int i = 0; i < 3; i++) {

  noTone(BUZZER_PIN);

  tone(BUZZER_PIN, 600, 80); delay(100);
  tone(BUZZER_PIN, 800, 80); delay(100);
  tone(BUZZER_PIN, 1000, 80); delay(100);
  noTone(BUZZER_PIN);
}

void somErro() {
  tone(BUZZER_PIN, 200, 300);
  delay(320);
  noTone(BUZZER_PIN);
}

void somTecla() {
  tone(BUZZER_PIN, 1200, 30);
  delay(40);
  noTone(BUZZER_PIN);
}

// --- LEDs ---

  digitalWrite(LED_VERDE, wifiOk ? HIGH : LOW);
  digitalWrite(LED_VERMELHO, wifiOk ? LOW : HIGH);
}

// ==================== LCD ====================
void lcdLimpar() { lcd.clear(); lcd.setCursor(0, 0); }

void lcdCentralizado(int linha, const char* texto) {
  int len = strlen(texto);
  int col = max(0, (16 - len) / 2);
  lcd.setCursor(col, linha);
  lcd.print(texto);
}

void mostrarEspera() {
  lcdLimpar();
  lcdCentralizado(0, "Digite a");
  lcdCentralizado(1, "proxima senha");
}

void mostrarDigitacao() {
  lcdLimpar();
  lcd.setCursor(0, 0);
  lcd.print("Senha: ");
  lcd.print(bufferDigita);
  for (int i = posDigita; i < MAX_DIGITOS; i++) lcd.print("_");
  lcdCentralizado(1, "C=OK  A=Apaga D=Limpa");
}

void mostrarConfirmacao() {
  lcdLimpar();
  lcdCentralizado(0, "Tem certeza?");
  lcdCentralizado(1, "C=Sim  A=Nao");
}

void mostrarSucesso(int senha) {
  lcdLimpar();
  lcdCentralizado(0, "Senha registrada!");
  char buf[17];
  sprintf(buf, "Senha: %04d", senha);
  lcdCentralizado(1, buf);
}

void mostrarErro(const char* msg) {
  lcdLimpar();
  lcdCentralizado(0, "ERRO");
  lcdCentralizado(1, msg);
}

void mostrarIP() {
  lcdLimpar();
  lcdCentralizado(0, "IP do ESP32:");
  lcdCentralizado(1, WiFi.localIP().toString().c_str());
}

// ==================== EEPROM ====================
void salvarEstado() {
  EEPROM.writeInt(ADDR_SENHA_ATUAL, estado.senhaAtual);
  EEPROM.writeInt(ADDR_TOTAL_HIST, estado.totalHistorico);
  for (int i = 0; i < LIMITE_HISTORICO; i++) {
    EEPROM.put(ADDR_HISTORICO + i * sizeof(RegistroSenha), estado.historico[i]);
  }
  EEPROM.commit();
}

void carregarEstado() {
  EEPROM.begin(EEPROM_SIZE);
  estado.senhaAtual = EEPROM.readInt(ADDR_SENHA_ATUAL);
  estado.totalHistorico = EEPROM.readInt(ADDR_TOTAL_HIST);

  if (estado.senhaAtual < MIN_SENHA || estado.senhaAtual > MAX_SENHA) estado.senhaAtual = 0;
  if (estado.totalHistorico < 0 || estado.totalHistorico > LIMITE_HISTORICO) estado.totalHistorico = 0;

  for (int i = 0; i < LIMITE_HISTORICO; i++) {
    EEPROM.get(ADDR_HISTORICO + i * sizeof(RegistroSenha), estado.historico[i]);
    if (!estado.historico[i].valido) estado.historico[i] = {0, "", false};
  }
}

// ==================== LÓGICA DE SENHAS ====================
void adicionarHistorico(int senha) {
  if (estado.senhaAtual > 0) {
    for (int i = LIMITE_HISTORICO - 1; i > 0; i--) {
      estado.historico[i] = estado.historico[i - 1];
    }
    estado.historico[0] = {estado.senhaAtual, "", true};
    strcpy(estado.historico[0].horario, obterHorario().c_str());
    if (estado.totalHistorico < LIMITE_HISTORICO) estado.totalHistorico++;
  }
  estado.senhaAtual = senha;
  salvarEstado();
}

void chamarProxima() {
  int prox = estado.senhaAtual + 1;
  if (prox > MAX_SENHA) prox = MIN_SENHA;
  adicionarHistorico(prox);
  somChamada();
  somConfirmacao();
  mostrarSucesso(prox);
  delay(2000);
  mostrarEspera();
}

void resetarContador() {
  estado.senhaAtual = 0;
  estado.totalHistorico = 0;
  for (int i = 0; i < LIMITE_HISTORICO; i++) estado.historico[i] = {0, "", false};
  salvarEstado();
  somConfirmacao();
  lcdLimpar();
  lcdCentralizado(0, "Contador resetado");
  delay(1500);
  mostrarEspera();
}

void definirSenhaManual(int senha) {
  if (senha >= MIN_SENHA && senha <= MAX_SENHA) {
    adicionarHistorico(senha);
    somConfirmacao();
    mostrarSucesso(senha);
    delay(2000);
  } else {
    somErro();
    mostrarErro("Senha invalida");
    delay(1500);
  }
  mostrarEspera();
}

// ==================== SERVIDOR WEB ====================
void handleRoot() {
  String html = FPSTR(HTML_INDEX);
  server.send(200, "text/html", html);
}

void addCORS() {
  server.sendHeader("Access-Control-Allow-Origin", "*");
  server.sendHeader("Access-Control-Allow-Methods", "GET, POST, OPTIONS");
  server.sendHeader("Access-Control-Allow-Headers", "Content-Type");
}

void handleApiSenha() {
  addCORS();
  JsonDocument doc;
  doc["senha"] = estado.senhaAtual > 0 ? String(estado.senhaAtual) : "---";
  doc["horario"] = obterHorario();
  doc["online"] = true;
  doc["historico_count"] = estado.totalHistorico;
  String resp; serializeJson(doc, resp);
  server.send(200, "application/json", resp);
}

void handleApiHistorico() {
  addCORS();
  JsonDocument doc;
  JsonArray arr = doc["historico"].to<JsonArray>();
  for (int i = 0; i < estado.totalHistorico; i++) {
    JsonObject obj = arr.add<JsonObject>();
    obj["senha"] = estado.historico[i].numero;
    obj["horario"] = estado.historico[i].horario;
  }
  String resp; serializeJson(doc, resp);
  server.send(200, "application/json", resp);
}

void handleApiChamar() {
  addCORS();
  chamarProxima();
  broadcastEstado();
  handleApiSenha();
}

void handleApiReset() {
  addCORS();
  resetarContador();
  broadcastEstado();
  handleApiSenha();
}

void handleApiSet() {
  addCORS();
  if (server.hasArg("senha")) {
    int num = server.arg("senha").toInt();
    definirSenhaManual(num);
  }
  broadcastEstado();
  handleApiSenha();
}

void handleOptions() {
  addCORS();
  server.send(204);
}

void handleNotFound() {
  addCORS();
  server.send(404, "text/plain", "Nao encontrado");
}

// ==================== WEBSOCKET BROADCAST ====================
void broadcastEstado() {
  JsonDocument doc;
  doc["senha"] = estado.senhaAtual > 0 ? String(estado.senhaAtual) : "---";
  doc["horario"] = obterHorario();
  doc["historico_count"] = estado.totalHistorico;
  JsonArray arr = doc["historico"].to<JsonArray>();
  for (int i = 0; i < estado.totalHistorico; i++) {
    JsonObject obj = arr.add<JsonObject>();
    obj["senha"] = estado.historico[i].numero;
    obj["horario"] = estado.historico[i].horario;
  }
  String resp; serializeJson(doc, resp);
  webSocket.broadcastTXT(resp);
}

// ==================== TECLADO - NOVA LÓGICA ====================
// A = APAGAR (backspace)
// B = CANCELAR / VOLTAR
// C = CONFIRMAR (enter/ok)
// D = MODO MANUAL / LIMPAR TUDO
// * = CHAMAR PRÓXIMA SENHA (atalho rápido)
// # = MOSTRAR IP DO ESP32

void processarTecla(char tecla) {
  ultimaTecla = millis();
  somTecla();

  Serial.printf("Tecla: %c | ModoManual: %d | ModoConfirm: %d | Buffer: %s\n",
                tecla, modoManual, modoConfirmacao, bufferDigita);

  // --- AÇÕES GLOBAIS (funcionam em qualquer modo) ---
  if (tecla == '*') {  // * = CHAMAR PRÓXIMA
    chamarProxima();
    broadcastEstado();
    return;
  }
  if (tecla == '#') {  // # = MOSTRAR IP
    mostrarIP();
    delay(3000);
    if (modoManual) mostrarDigitacao();
    else if (modoConfirmacao) mostrarConfirmacao();
    else mostrarEspera();
    return;
  }

  // --- MODO CONFIRMAÇÃO ---
  if (modoConfirmacao) {
    if (tecla == 'C') {  // C = CONFIRMAR = SIM
      int senha = atoi(bufferDigita);
      if (senha >= MIN_SENHA && senha <= MAX_SENHA) {
        definirSenhaManual(senha);
        broadcastEstado();
      } else {
        somErro();
        mostrarErro("Senha invalida");
        delay(1500);
        mostrarEspera();
      }
      limparBuffer();
      modoConfirmacao = false;
      modoManual = false;
    }
    else if (tecla == 'A' || tecla == 'B') {  // A/B = NÃO / CANCELAR
      modoConfirmacao = false;
      mostrarDigitacao();
    }
    return;
  }

  // --- MODO DIGITAÇÃO MANUAL ---
  if (modoManual) {
    if (tecla >= '0' && tecla <= '9') {
      if (posDigita < MAX_DIGITOS) {
        bufferDigita[posDigita++] = tecla;
        bufferDigita[posDigita] = '\0';
      }
    }
    else if (tecla == 'A') {  // A = APAGAR (backspace)
      if (posDigita > 0) {
        bufferDigita[--posDigita] = '\0';
      }
    }
    else if (tecla == 'D') {  // D = LIMPAR TUDO
      limparBuffer();
    }
    else if (tecla == 'C') {  // C = CONFIRMAR
      if (posDigita > 0) {
        modoConfirmacao = true;
        mostrarConfirmacao();
      }
    }
    else if (tecla == 'B') {  // B = SAIR modo manual
      modoManual = false;
      limparBuffer();
      mostrarEspera();
      return;
    }
    mostrarDigitacao();
    return;
  }

  // --- MODO ESPERA (inicial) ---
  // D = Entrar modo digitação manual
  if (tecla == 'D') {
    modoManual = true;
    limparBuffer();
    mostrarDigitacao();
  }
  // Outras teclas ignoradas no modo espera
}

void limparBuffer() {
  memset(bufferDigita, 0, sizeof(bufferDigita));
  posDigita = 0;
}

// ==================== WIFI ====================
void conectarWiFi() {
  WiFi.mode(WIFI_STA);
  WiFi.begin(WIFI_SSID, WIFI_PASSWORD);

  lcdLimpar();
  lcdCentralizado(0, "Conectando WiFi");
  lcd.setCursor(0, 1);
  lcd.print(WIFI_SSID);

  int tentativas = 0;
  while (WiFi.status() != WL_CONNECTED && tentativas < 30) {
    delay(500);
    lcd.print(".");
    tentativas++;
  }

  if (WiFi.status() == WL_CONNECTED) {
    lcdLimpar();
    lcdCentralizado(0, "WiFi Conectado!");
    lcdCentralizado(1, WiFi.localIP().toString().c_str());
    atualizarLeds(true);
    delay(2000);
  } else {
    lcdLimpar();
    lcdCentralizado(0, "Falha WiFi");
    lcdCentralizado(1, "Modo Offline");
    atualizarLeds(false);
    delay(2000);
  }
}

// ==================== SETUP ====================
void setup() {
  Serial.begin(115200);
  delay(100);
  Serial.println("\n=== Sistema de Senhas ESP32 ===");
  Serial.println("Keypad: *=Proxima  #=IP  A=Apagar  B=Cancelar  C=Confirmar  D=Manual");

  // Pinos
  pinMode(BUZZER_PIN, OUTPUT);
  pinMode(LED_VERDE, OUTPUT);
  pinMode(LED_VERMELHO, OUTPUT);
  digitalWrite(LED_VERDE, LOW);
  digitalWrite(LED_VERMELHO, HIGH);
  noTone(BUZZER_PIN);

  // Keypad
  keypad.setDebounceTime(50);

  // LCD I2C (SDA=22, SCL=23) - Auto-detecta endereço 0x27 ou 0x3F
  Wire.begin(22, 23);
  if (!iniciarLCD()) {
    Serial.println("AVISO: LCD nao encontrado. Continuando sem display...");
  }
  lcdLimpar();
  lcdCentralizado(0, "Iniciando...");
  lcdCentralizado(1, "v1.0");
  delay(1000);

  // EEPROM
  carregarEstado();

  // WiFi
  conectarWiFi();

  // Servidor Web
  server.on("/", handleRoot);
  server.on("/api/senha", handleApiSenha);
  server.on("/api/historico", handleApiHistorico);
  server.on("/api/chamar", HTTP_POST, handleApiChamar);
  server.on("/api/reset", HTTP_POST, handleApiReset);
  server.on("/api/set", HTTP_POST, handleApiSet);
  server.on("/", HTTP_OPTIONS, handleOptions);
  server.on("/api/senha", HTTP_OPTIONS, handleOptions);
  server.on("/api/historico", HTTP_OPTIONS, handleOptions);
  server.on("/api/chamar", HTTP_OPTIONS, handleOptions);
  server.on("/api/reset", HTTP_OPTIONS, handleOptions);
  server.on("/api/set", HTTP_OPTIONS, handleOptions);
  server.onNotFound(handleNotFound);
  server.begin();

  // WebSocket
  webSocket.begin();
  webSocket.onEvent(webSocketEvent);

  // Configura tempo real (NTP) se WiFi conectado
  if (WiFi.status() == WL_CONNECTED) {
    configTime(-3 * 3600, 0, "pool.ntp.org", "time.nist.gov"); // UTC-3 Brasil
  }

  // Inicialização
  somConfirmacao();
  delay(100);
  somConfirmacao();

  mostrarEspera();

  Serial.printf("Sistema pronto! IP: %s\n", WiFi.localIP().toString().c_str());
  Serial.printf("WebSocket: ws://%s:81\n", WiFi.localIP().toString().c_str());
}

// ==================== LOOP ====================
void loop() {
  server.handleClient();
  webSocket.loop();

  char tecla = keypad.getKey();
  if (tecla) processarTecla(tecla);

  // Timeout modo manual (30s)
  if (modoManual && millis() - ultimaTecla > 30000) {
    modoManual = false;
    modoConfirmacao = false;
    limparBuffer();
    mostrarEspera();
  }

  // Timeout confirmação (10s)
  if (modoConfirmacao && millis() - ultimaTecla > 10000) {
    modoConfirmacao = false;
    if (modoManual) mostrarDigitacao();
    else mostrarEspera();
  }

  // Reconexão WiFi
  static unsigned long ultimaVerificacao = 0;
  if (millis() - ultimaVerificacao > 30000) {
    ultimaVerificacao = millis();
    if (WiFi.status() != WL_CONNECTED) {
      Serial.println("WiFi caiu, reconectando...");
      atualizarLeds(false);
      conectarWiFi();
    }
  }

  delay(10);
}

# Sistema de Senhas - Lanchonete (Wokwi Simulation)

Projeto ESP32 para sistema de senhas de atendimento em lanchonete, pronto para simulação no [Wokwi](https://wokwi.com).

## 🎯 Funcionalidades

- **LCD 16x2 I2C** - Exibe mensagens e senhas
- **Keypad 4x4** - Entrada de senhas e comandos
- **Buzzer** - Sons de confirmação, erro e chamada
- **LEDs** - Verde (WiFi OK) / Vermelho (WiFi falhou)
- **WebServer** - Painel web acessível via navegador
- **WebSocket** - Atualização em tempo real
- **EEPROM** - Persistência de senhas entre reinícios

## 🔌 Conexões (conforme diagram.json)

| Componente | Pino ESP32 |
|------------|------------|
| LCD SDA    | GPIO 22    |
| LCD SCL    | GPIO 23    |
| Keypad R1  | GPIO 13    |
| Keypad R2  | GPIO 12    |
| Keypad R3  | GPIO 14    |
| Keypad R4  | GPIO 27    |
| Keypad C1  | GPIO 26    |
| Keypad C2  | GPIO 25    |
| Keypad C3  | GPIO 33    |
| Keypad C4  | GPIO 32    |
| Buzzer     | GPIO 18    |
| LED Verde  | GPIO 19    |
| LED Vermelho | GPIO 21  |

## ⌨️ Layout do Keypad

```
1  2  3  A     <- A = APAGAR (backspace)
4  5  6  B     <- B = CANCELAR / VOLTAR
7  8  9  C     <- C = CONFIRMAR (enter/ok)
*  0  #  D     <- * = CHAMAR PRÓXIMA, # = MOSTRAR IP, D = MODO MANUAL
```

## 📋 Funcionamento

### Modo Espera (inicial)
- Pressione **D** para entrar no modo digitação manual
- Pressione **\*** para chamar próxima senha automaticamente
- Pressione **#** para ver o IP do ESP32

### Modo Digitação Manual
- Digite números **0-9** (máx 4 dígitos)
- **A** = Apagar último dígito
- **D** = Limpar tudo
- **C** = Confirmar (vai para modo confirmação)
- **B** = Cancelar e voltar ao modo espera

### Modo Confirmação
- **C** = Sim, confirmar senha digitada
- **A** ou **B** = Não, voltar para digitação

## 🔧 Configuração WiFi

As credenciais estão no topo do `sketch.ino` (linhas 31-32):

```cpp
const char* WIFI_SSID = "ESP-PROJETO";
```

- **Para hardware real**: Edite `WIFI_SSID` e `WIFI_PASSWORD` com suas credenciais
- **Para Wokwi online**: Mude temporariamente para `"Wokwi-GUEST"` (rede simulada aberta)
